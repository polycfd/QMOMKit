add_executable(test_distributions_analytical distributions_analytical.cpp)
target_include_directories(test_distributions_analytical PRIVATE "${QMOMKIT_REPOSITORY_ROOT}/src")
target_link_libraries(test_distributions_analytical PRIVATE qmomkit)
foreach(case_name IN ITEMS families mixtures independent_products configuration_initialization)
    if(case_name STREQUAL configuration_initialization)
        set(test_name verification_distribution_configuration_initialization)
    else()
        set(test_name verification_distribution_${case_name}_analytical)
    endif()
    add_test(NAME ${test_name} COMMAND test_distributions_analytical ${case_name})
    set_tests_properties(${test_name} PROPERTIES LABELS verification)
endforeach()
